\section{Context and Motivation}

\begin{frame}{The Hennessy--Patterson processor}{Context}
    \begin{columns}[T]
    \begin{column}{0.4\textwidth}
        \begin{itemize}
            \item \textit{Computer Organization and Design: RISC-V Edition} is
                  the reference in computer architecture courses.
            \item Used in Digital Systems II (PCS3225) at Poli.
            \item It implements a single-cycle processor with 8 instructions:
                  \texttt{lw sw beq add addi sub and or}
        \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
        \includegraphics[width=\textwidth]{hannersy-patterson-implementation-fig4-21.png}

        \vspace{0.3em}
        \tiny Source: Patterson \& Hennessy, \textit{Computer Organization and
        Design: RISC-V Edition}, 2nd ed., 2020, Figure 4.21.
    \end{column}
    \end{columns}
\end{frame}
% SAY:
% - The Patterson and Hennessy book is the course reference.
% - Students implement exactly this datapath (Figure 4.21) in VHDL.
% - Eight instructions: two memory accesses, one branch, five ALU operations.
% - No lui, no jalr: no large addresses, and no function calls.

\begin{frame}{Problem}{Motivation}
    \vspace{0.5em}
    {\centering\Large\color{PoliBlue}
     There is no compiler for only 8 instructions.\par}
    \vspace{1em}
    \begin{itemize}
        \item GCC and Clang emit at least 40 instructions (RV32I).
        \item Students and instructors have to write programs by hand in
              assembly, which caps how large those programs can get.
        \item The chance to build on the C the students already know, and to
              introduce compiler translation and optimization, is lost.
    \end{itemize}
\end{frame}
% SAY:
% - The mismatch is simple: even the most restricted compiler emits far more
%   than the student's hardware understands.
% - So the lab stays stuck with hand-written assembly and toy programs.
% - And the hardware-software link, which is the pedagogical point, never closes.

\begin{frame}{Objective}
    Generate assembly from C for progressively larger instruction sets,
    synthesizing every missing instruction with an equivalent sequence
    inside GCC.

    \vspace{1em}
    \centering\small
    \begin{tabular}{ll}
        \toprule
        \textbf{Target} & \textbf{ISA} \\
        \midrule
        rvsc0 & \texttt{lw sw beq add addi sub and or} \\
        rvsc1 & + \texttt{lui}, \texttt{jalr} \\
        rvsc2 & RV32I $-$ fence $-$ Zicsr \\
        rvsc3 & full RV32I \\
        rvsc4 & RV64I \\
        rvsc5 & RV64IM \\
        rvsc6 & RV64IMFD + Zicsr \\
        rvsc7 & RV64IMAFD \\
        \bottomrule
    \end{tabular}
\end{frame}
% SAY:
% - Eight targets, each adding one group of instructions.
% - rvsc0 is exactly the processor from the book.
% - rvsc1 adds lui and jalr: the first target with function calls, and
%   therefore the first that compiles a real C program.
% - From rvsc3 on it is standard RISC-V; the interest is in the first two.
% - The progression follows how the processor evolves during the course.

\section{Related Work}

\begin{frame}{Related work}
    \begin{itemize}
        \item \textbf{Berkeley SoftFloat} --- floating point in software.
        \item \textbf{MSP430}: \texttt{\_\_mspabi\_slli\_n} --- multi-bit shift
              on a single-shift ISA.
        \item \textbf{AVR}: \texttt{\_\_mulhi3} --- multiplication in software.
    \end{itemize}
\end{frame}
% SAY:
% - Compiling for incomplete hardware is not new: no FPU, no multiplier.
% - The classic answer is a library routine, libgcc.
% - That does not work here for two reasons: rvsc0 cannot even make a call,
%   and a call hides the real cost of the instruction from the student.
% - All of these are solved in a library, without touching the compiler.
% - Here the synthesis lives in the machine description, inline.

\section{Development}

\begin{frame}{GCC pipeline}
    \begin{center}
        \includegraphics[width=0.9\textwidth]{gcc-internal-flow.jpg}

        \vspace{0.2em}
        \tiny Source: AlexeySmirnov, \textit{GNU C Compiler Internals}
        (Wikibooks), file \texttt{Gcc.JPG}, CC BY-SA 3.0.
    \end{center}
\end{frame}
% SAY:
% - This is the GCC pipeline, and only one point matters: expand.
% - Expand is the moment GCC asks the machine description
%   "how do I do an xor here?".
% - If I answer with three instructions instead of one, the xor stops existing.
% - Doing it at expand, and not later, is what gives correctness by
%   construction: no optimization pass can bring back what was never there.

\begin{frame}[fragile]{Example: XOR}
    \lstset{basicstyle=\ttfamily\normalsize}
    \large
    \textbf{Source}
\begin{lstlisting}[language=C]
int f(int a, int b) { return a ^ b; }
\end{lstlisting}
    \textbf{GIMPLE}
\begin{lstlisting}[language=C]
D.2418 = a ^ b;
return D.2418;
\end{lstlisting}
    \textbf{RTL}
\begin{lstlisting}[language=lisp]
(set (reg:SI 137) (xor:SI (reg:SI 135) (reg:SI 136)))
\end{lstlisting}
    \textbf{Assembly}
\begin{lstlisting}
f:
    xor  a0, a0, a1
    ret
\end{lstlisting}
\end{frame}

% SAY:
% - First the normal path, with TARGET_XOR enabled.
% - Expand does nothing: the define_insn matches directly and an xor comes out.
% - Keep an eye on the RTL in the middle: that is what changes next slide.

\begin{frame}[fragile]{Example: "soft" XOR}
    \lstset{basicstyle=\ttfamily\normalsize}
    \large
    \textbf{RTL}
\begin{lstlisting}[language=lisp]
(set (reg 138) (and:SI (reg 135) (reg 136)))
(set (reg 139) (ior:SI (reg 135) (reg 136)))
(set (reg 137) (minus:SI (reg 139) (reg 138)))
   (REG_EQUAL (xor:SI ...))
\end{lstlisting}
    \textbf{Assembly}
\begin{lstlisting}
f:
    and  a5, a0, a1
    or   a0, a0, a1
    sub  a0, a0, a5
    ret
\end{lstlisting}
\end{frame}
% SAY:
% - Same source, same GIMPLE. The only difference is the -mno-xor flag, which
%   the target header injects by itself through CC1_SPEC.
% - Expand sees that xor does not exist, emits the identity and calls DONE.
% - Real cost: 3 instructions and 1 extra register.
% - There are two pseudo-registers (138 and 139), not one reused: that is the
%   dataflow edge that stops the allocator from clobbering a|b.
% - REG_EQUAL preserves the semantics for CSE and combine without letting
%   either of them reintroduce the xor.
% - The two pseudos are not cosmetic: the earlier De Morgan form reused a
%   register, IRA aliased it and the code came out wrong. That was a real bug,
%   caught by the behavioral tests.

\begin{frame}[fragile]{Example: SLL}
    \vspace{0.5em}
    {\centering\large\color{PoliBlue}
     $\mathrm{sll}(x,0) = x$

     \vspace{0.4em}
     $\mathrm{sll}(x,n{+}1) = \mathrm{sll}(x,n) + \mathrm{sll}(x,n)$\par}
    \vspace{1.2em}

    \lstset{basicstyle=\ttfamily\normalsize}
    \large
    \begin{itemize}
        \item If $n$ is immediate:
\begin{lstlisting}
add rd, rd, rd
... # n times
\end{lstlisting}
        \item If $n$ is a parameter:
\begin{lstlisting}[language=C]
for (uint32_t rd = rs1, i = 0; i < rs2; i++) rd += rd;
\end{lstlisting}
    \end{itemize}
\end{frame}
% SAY:
% - Every synthesis comes from an algebraic derivation, not from trial and error.
% - Left shift is the cleanest case: by induction, it is adding a value to itself.
% - With a constant count the cost is trivial and the code is straight-line.
% - With a variable count, instead of a counting-down loop, I decompose the
%   count in binary: five bit tests cover any value from 0 to 31.
% - That took the worst case from 187 down to 47 instructions.

\begin{frame}{Theoretical dynamic cost}
    \centering\scriptsize
    \begin{tabular}{lrr}
        \toprule
        \textbf{Operation} & \textbf{Worst case} & \textbf{Extra regs} \\
        \midrule
        \texttt{not}                       & 2        & 0 \\
        \texttt{xor} (reg / imm)           & 3 / 4    & 1 / 2 \\
        \texttt{ori} / \texttt{andi}       & 2        & 1 \\
        \texttt{sll} const / var           & 31 / 47  & 0 / 1 \\
        \texttt{srl} const / var           & 158 / 332 & 3 \\
        \texttt{sra} const / var           & 194 / 391 & 5 / 7 \\
        \texttt{slt} / \texttt{sltu}       & 49 / 48  & 3 / 4 \\
        \texttt{bne}                       & 3        & 1 \\
        \texttt{blt},\texttt{bltu} / \texttt{bge},\texttt{bgeu} & 16 / 17 & 7 \\
        \texttt{lb}/\texttt{lbu}           & 77 / 460 & 2 \\
        \texttt{lh}/\texttt{lhu}           & 85 / 532 & 2 \\
        \texttt{sb}                        & 32 / 95  & 3 \\
        \texttt{sh}                        & 23 / 82  & 3 \\
        \texttt{lui} (rvsc0)               & 25       & 0 \\
        \texttt{jal} (rvsc1)               & 5        & 1 \\
        \bottomrule
    \end{tabular}
\end{frame}
% SAY:
% - These are measured numbers, not estimates: I run a loop with the operation
%   and an empty loop on the simulator and subtract, so the overhead cancels
%   exactly.
% - Two readings matter.
% - First: the cost spans three orders of magnitude. NOT costs 2; a byte load
%   with an unknown address costs 460.
% - Second: "known lane" versus "unknown lane" are different sequences, not a
%   range. If the compiler knows which byte of the word the datum sits in
%   (a global, a local, a struct field), the whole address arithmetic
%   disappears. That was the single most profitable optimization.

\section{Validation}

\begin{frame}{Testing}{Methodology}
    \begin{itemize}
        \item \textbf{Static validation:} \texttt{objdump -M no-aliases} and a
              check of every generated mnemonic against the allowed list.
        \item \textbf{Behaviour validation:} programs that validate themselves
              through \texttt{assert} in their own code, executed on the
              Spike emulator.
    \end{itemize}
    \vfill
\end{frame}

\begin{frame}{Validation base}{Methodology}
    \begin{itemize}
        \item \textbf{rvsc0}
        \begin{itemize}
            \item 72 hand-written smoke tests.
        \end{itemize}
        \item \textbf{rvsc1}
        \begin{itemize}
            \item 48 hand-written smoke tests.
            \item GCC torture suite: 1684 programs used by GCC itself.
        \end{itemize}
        \item \textbf{rvsc2}
        \begin{itemize}
            \item 3 hand-written smoke tests.
        \end{itemize}
    \end{itemize}
    \vfill
    \centering Every test at 5 optimization levels
    (\texttt{-O0} \dots{} \texttt{-Os}).
\end{frame}

% SAY:
% - Validation answers two different questions, which is why it is layered.
% - "Correct by construction": no forbidden mnemonic was emitted. That is static.
% - "Correct by execution": the computed value is what C requires. That runs.
% - The torture suite is what gives scale: 1684 programs I did not write.
% - Checking the linked ELF matters because the static check only looks at what
%   the compiler produced; the library shipped hand-written assembly with
%   forbidden instructions, and only that check caught it.

\begin{frame}{Torture suite (rvsc1)}{Result}
    \begin{itemize}
        \item 13 tests had to be skipped:
        \begin{itemize}
            \item \texttt{run\_expensive\_tests} (timeout)
            \item \texttt{c99\_runtime}
            \item \texttt{int128}
            \item \texttt{mmap} and \texttt{dfp}
        \end{itemize}
        \item \textbf{1669 tests passed} emitting only the 10 sc1
              instructions, for a total of \textbf{8345 runs}.
    \end{itemize}
\end{frame}
% SAY:
% - This is the number that gives confidence in the work: eight thousand three
%   hundred program/optimization-level combinations, zero failures.
% - The delicate part is the skips: a skip can hide a failure.
% - So the skip list is derived, not curated: I read the test's own dg-require
%   directive, the same rule DejaGnu applies.
% - And I confirmed by comparison that a plain rv32i GCC fails exactly the same
%   cases, so none of them is the restricted ISA's fault.
% - Zero ICEs means the compiler never crashed.

% ==================================================================
\begin{frame}{Embench-IoT benchmark}{Result}
    \begin{itemize}
        \item Free benchmark suite for \textbf{IoT / embedded} --- small, real
              programs with no operating system dependency.
        \item \textbf{19 cases}: cryptography (\texttt{nettle-aes},
              \texttt{nettle-sha256}, \texttt{md5sum}, \texttt{crc32}),
              signal and image processing (\texttt{edn},
              \texttt{picojpeg}, \texttt{depthconv}, \texttt{qrduino}),
              data structures and strings (\texttt{sglib-combined},
              \texttt{slre}, \texttt{wikisort}, \texttt{huffbench},
              \texttt{tarfind}), arithmetic and control
              (\texttt{aha-mont64}, \texttt{matmult-int}, \texttt{ud},
              \texttt{xgboost}, \texttt{statemate}, \texttt{nsichneu}).
        \item Two independent measures: \textbf{size} (\texttt{.text} at
              \texttt{-O2}) and \textbf{time} (retired instructions on Spike).
        \item Two toolchains: \textbf{\texttt{riscv32-unknown-elf}} / \textbf{rvsc1-unknown-elf} (synthesized).
    \end{itemize}
\end{frame}

\section{Conclusion}

\begin{frame}{Contributions}{Conclusion}
    \small
    \begin{itemize}
        \item The objectives were met:
        \begin{itemize}
            \item \texttt{rvsc0} --- eight instructions.
            \item \texttt{rvsc1} --- a full C calling convention with ten instructions.
            \item \texttt{rvsc2}--\texttt{rvsc7} --- RV32I $\rightarrow$ RV64IMAFD.
        \end{itemize}
        \item Derivations and optimizations were defined for 31 different
              instructions.
        \item A validation framework was built on top of already existing test
              suites.
        \item \textbf{Cost:} $\times 6.99$ static, $\times 15.8$ dynamic,
              an acceptable figure for an introductory course.
        \end{itemize}
\end{frame}
